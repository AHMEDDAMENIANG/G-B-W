// Connexion automatique et récupération des produits depuis Supabase async function loadProducts() { const loadingMessage = document.getElementById("loadingMessage"); const productsGrid = document.getElementById("productsGrid"); try { const { data: products, error } = await supabase .from('products') .select('*'); if (error) throw error; if (!products || products.length === 0) { loadingMessage.textContent = "Aucun produit n'est disponible pour le moment."; return; } loadingMessage.style.display = "none"; productsGrid.innerHTML = ""; // Vide la grille avant d'afficher products.forEach(product => { const productCard = document.createElement("div"); productCard.className = "product-card"; const imageUrl = product.image_url || 'https://placeholder.com'; const displayPrice = product.selling_price || product.price; // Échappement simple pour éviter les bugs avec les apostrophes dans les noms de produits const safeName = product.name.replace(/'/g, "\\'"); productCard.innerHTML = `

${product.name}

${product.description || 'Aucune description disponible.'}

${displayPrice} FCFA
`; productsGrid.appendChild(productCard); }); } catch (error) { loadingMessage.textContent = "Erreur lors du chargement : " + error.message; console.error(error); } } // FONCTION PANIER : Mémorise l'article localement sur le navigateur function addToCart(id, name, price) { let cart = JSON.parse(localStorage.getItem('gbw_cart')) || []; const existingItem = cart.find(item => item.id === id); if (existingItem) { existingItem.quantity += 1; } else { cart.push({ id: id, name: name, price: price, quantity: 1 }); } localStorage.setItem('gbw_cart', JSON.stringify(cart)); alert(`${name} a été ajouté à votre panier ! 🛒`); } // Lancer le chargement au démarrage de la page loadProducts();
